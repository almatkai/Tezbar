   if std::env::var("SECRET_MARKER").is_ok(){panic!("ambient secret inherited")}

  println!("{{\"jsonrpc\":\"2.0\",\"id\":\"{id}\",\"result\":{result}}}");io::stdout().flush().unwrap();
 }
}
